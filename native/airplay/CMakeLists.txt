cmake_minimum_required(VERSION 3.13)
# klm-airplay — penerima AirPlay Mirroring untuk KLM Mix. Memakai pustaka AirPlay dari UxPlay (GPL-3.0) → program ini GPL-3.0.
#   git clone https://github.com/FDH2/UxPlay.git
#   cmake -S native/airplay -B native/airplay/build -DUXPLAY_DIR=<path UxPlay> -G Ninja && cmake --build native/airplay/build
# Butuh: OpenSSL (libcrypto) dan libplist-2.0. Di Windows dibangun dengan MSYS2 (UCRT64).
project(klm_airplay C)
set(UXPLAY_DIR "" CACHE PATH "Path sumber UxPlay")
if(NOT EXISTS "${UXPLAY_DIR}/lib/raop.h")
  message(FATAL_ERROR "UXPLAY_DIR tidak valid: ${UXPLAY_DIR}")
endif()
set(CMAKE_C_STANDARD 11)
set(NO_MARCH_NATIVE ON CACHE BOOL "" FORCE)   # biner harus jalan di semua PC, bukan hanya mesin build
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# mDNS bawaan UxPlay (tidak perlu Bonjour), ditambal sedikit: alamat jaringan bisa dipilih lewat env KLM_MDNS_IFACE
# sehingga KLM Mix bisa menjalankan satu penerima per jaringan (WiFi, LAN, Hotspot Seluler PC) — bukan hanya rute bawaan.
file(READ ${UXPLAY_DIR}/lib/mdnsd/mdnsd.c MDNSD_SRC)
set(MDNSD_OLD "static uint32_t mdns_get_default_ipv4(void)\n{\n    uint32_t addr = 0;")
set(MDNSD_NEW "static uint32_t mdns_get_default_ipv4(void)\n{\n    uint32_t addr = 0;\n    { const char *ov = getenv(\"KLM_MDNS_IFACE\"); struct in_addr ova; if (ov && *ov && inet_pton(AF_INET, ov, &ova) == 1) return ova.s_addr; }")
string(REPLACE "${MDNSD_OLD}" "${MDNSD_NEW}" MDNSD_PATCHED "${MDNSD_SRC}")
if(MDNSD_PATCHED STREQUAL MDNSD_SRC)
  message(FATAL_ERROR "tambalan mdnsd tidak cocok dengan versi UxPlay ini")
endif()
string(REPLACE "#include \"../compat.h\"" "#include \"compat.h\"" MDNSD_PATCHED "${MDNSD_PATCHED}")
file(WRITE ${CMAKE_BINARY_DIR}/mdnsd_klm/mdnsd.c "${MDNSD_PATCHED}")
add_library(dnssd STATIC ${CMAKE_BINARY_DIR}/mdnsd_klm/mdnsd.c ${UXPLAY_DIR}/lib/mdnsd/dnssd_mdnsd.c)
target_include_directories(dnssd PRIVATE ${UXPLAY_DIR}/lib/mdnsd ${UXPLAY_DIR}/lib)
add_subdirectory(${UXPLAY_DIR}/lib/llhttp ${CMAKE_BINARY_DIR}/llhttp)
add_subdirectory(${UXPLAY_DIR}/lib/playfair ${CMAKE_BINARY_DIR}/playfair)
add_subdirectory(${UXPLAY_DIR}/lib ${CMAKE_BINARY_DIR}/airplay)
add_executable(klm-airplay klm-airplay.c)
target_include_directories(klm-airplay PRIVATE ${UXPLAY_DIR}/lib)
target_link_libraries(klm-airplay airplay)
# suara iPhone: libavcodec (FFmpeg, LGPL) — cukup dibangun minimal: --disable-everything --enable-decoder=aac,alac
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV QUIET libavcodec libavutil)
if(AV_FOUND)
  message(STATUS "klm-airplay: suara aktif (libavcodec ${AV_libavcodec_VERSION})")
  target_compile_definitions(klm-airplay PRIVATE HAVE_AVCODEC)
  target_include_directories(klm-airplay PRIVATE ${AV_INCLUDE_DIRS})
  target_link_directories(klm-airplay PRIVATE ${AV_STATIC_LIBRARY_DIRS})
  target_link_libraries(klm-airplay ${AV_STATIC_LIBRARIES})
else()
  message(WARNING "klm-airplay: libavcodec tidak ditemukan — dibangun TANPA suara")
endif()
if(WIN32)
  target_link_libraries(klm-airplay ws2_32 iphlpapi bcrypt)
else()
  target_link_options(klm-airplay PRIVATE -Wl,--no-as-needed)
  set_property(TARGET klm-airplay APPEND PROPERTY LINK_LIBRARIES -Wl,--start-group airplay m -Wl,--end-group)
endif()
